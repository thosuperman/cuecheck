cmake_minimum_required(VERSION 3.16)
project(CueCheck VERSION 0.1.0 LANGUAGES CXX)

add_executable(cuecheck src/main.cpp src/check.cpp)
target_compile_features(cuecheck PRIVATE cxx_std_17)
target_compile_definitions(cuecheck PRIVATE CUECHECK_VERSION="${PROJECT_VERSION}")
if(WIN32)
  enable_language(RC)
  configure_file(src/version.rc.in version.rc @ONLY)
  target_sources(cuecheck PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/version.rc")
endif()
if(MSVC)
  set_property(TARGET cuecheck PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(cuecheck PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
else()
  target_compile_options(cuecheck PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
  if(MINGW)
    target_link_options(cuecheck PRIVATE -municode -static)
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME cli COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/cli.py" "$<TARGET_FILE:cuecheck>")
endif()
